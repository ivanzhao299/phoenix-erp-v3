'use client';
import React, { useState } from 'react';
import { Upload, X, FileText } from 'lucide-react';

/**
 * AttachmentUploader — 统一多附件上传（基于现有 /api/upload/contract 通道）。
 * value: string[]（受控 fileId 或历史附件引用列表）；onChange 返回新列表。审批后 readOnly 只读。
 */
export interface AttachmentUploaderProps {
  value: string[];
  onChange: (urls: string[]) => void;
  readOnly?: boolean;
  label?: string;
  apiBase?: string;
}

export function AttachmentUploader({ value, onChange, readOnly, label = '上传附件（PDF/图片 ≤15MB）', apiBase }: AttachmentUploaderProps) {
  const [uploading, setUploading] = useState(false);
  const [err, setErr] = useState('');

  const token = (typeof window !== 'undefined'
    && (sessionStorage.getItem('phoenix_token') || localStorage.getItem('phoenix_token'))) || '';
  const base = apiBase ?? (process.env.NEXT_PUBLIC_API_BASE || '');

  // G134：支持一次多选 + 多次累加。批量内逐个串行上传，全部成功的 url 一次性合并进 value
  //   （避免并行 onChange 各读到 stale value 互相覆盖，导致"只剩最后一个"）。
  const MAX_MB = 15;   // 与应用层 FileInterceptor(15MB) + nginx(20m) 一致（原 10 与"≤15MB"标签不符）
  const uploadFiles = async (files: File[]) => {
    setErr(''); setUploading(true);
    const added: string[] = [];
    const errs: string[] = [];
    try {
      for (const file of files) {
        if (file.size > MAX_MB * 1024 * 1024) {
          errs.push(`${file.name}：超出 ${MAX_MB}MB（${(file.size / 1024 / 1024).toFixed(1)}MB）`);
          continue;
        }
        const fd = new FormData();
        fd.append('file', file);
        const res = await fetch(`${base}/api/upload/contract`, { method: 'POST', headers: token ? { Authorization: `Bearer ${token}` } : undefined, body: fd });
        // G30：5xx 返 text/HTML，不强 JSON.parse（防 "Unexpected token 'I'" 报错）
        const ct = res.headers.get('content-type') || '';
        if (!res.ok) {
          const detail = ct.includes('application/json') ? (await res.json().catch(() => ({})))?.message : await res.text().catch(() => '');
          errs.push(`${file.name}：上传失败 (${res.status})${detail ? '，' + String(detail).slice(0, 80) : ''}`);
          continue;
        }
        const json = ct.includes('application/json') ? await res.json() : {};
        if (json?.fileId) added.push(String(json.fileId));
        else if (json?.url) added.push(String(json.url));
        else errs.push(`${file.name}：服务端未返回受控 fileId`);
      }
      if (added.length) onChange([...value, ...added]);
      if (errs.length) setErr(errs.join('；'));
    } catch (e: any) { setErr(e?.message || '上传失败'); }
    finally { setUploading(false); }
  };

  return (
    <div>
      {value.length > 0 && (
        <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8, marginBottom: readOnly ? 0 : 8 }}>
          {value.map((u, i) => (
            <AttachmentChip key={i} url={u} index={i} readOnly={readOnly}
              onRemove={() => onChange(value.filter((_, idx) => idx !== i))} />
          ))}
        </div>
      )}
      {!readOnly && (
        <label style={{ display: 'inline-flex', alignItems: 'center', gap: 6, padding: '7px 14px', borderRadius: 'var(--radius-md)', border: '1px dashed var(--border)', color: 'var(--text-secondary)', cursor: uploading ? 'wait' : 'pointer', fontSize: 13 }}>
          <Upload size={14} />{uploading ? '上传中…' : label}
          {/* G134：multiple 允许一次选多个；也可重复点击逐个累加 */}
          <input type="file" multiple accept=".pdf,.jpg,.jpeg,.png,.webp" style={{ display: 'none' }} disabled={uploading}
            onChange={(e) => { const fs = Array.from(e.target.files ?? []); if (fs.length) uploadFiles(fs); e.target.value = ''; }} />
        </label>
      )}
      {readOnly && value.length === 0 && <span style={{ fontSize: 13, color: 'var(--text-disabled)' }}>无附件</span>}
      {err && <div style={{ fontSize: 12, color: 'var(--status-danger)', marginTop: 4 }}>{err}</div>}
    </div>
  );
}

function isUnsafeAttachmentRef(ref: string) {
  const value = ref.trim().toLowerCase();
  return value.startsWith('/uploads/') || value.startsWith('uploads/') || value.startsWith('http://') || value.startsWith('https://');
}

function isProtectedFileRef(ref: string) {
  return /^FILE-[A-Za-z0-9._:-]+$/.test(ref.trim());
}

// G169：已上传附件单项。新上传返回受控 fileId；历史裸 URL 只显示为禁用引用，不再渲染可点击裸链接。
function AttachmentChip({ url, index, readOnly, onRemove }: {
  url: string; index: number; readOnly?: boolean; onRemove: () => void;
}) {
  const unsafe = isUnsafeAttachmentRef(url);
  const protectedRef = isProtectedFileRef(url);
  const displayName = protectedRef ? url : `附件${index + 1}`;
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '5px 8px 5px 5px', background: 'var(--bg-muted)', border: '1px solid var(--border)', borderRadius: 'var(--radius-md)', fontSize: 12.5 }}>
      <FileText size={18} style={{ color: unsafe ? 'var(--status-warning)' : 'var(--text-secondary)' }} />
      <span
        title={unsafe ? '历史裸链接已被安全策略禁用' : (protectedRef ? '受控附件引用，需通过模块受控端点访问' : displayName)}
        style={{
          color: unsafe ? 'var(--status-warning)' : 'var(--color-accent)',
          maxWidth: 220,
          overflow: 'hidden',
          textOverflow: 'ellipsis',
          whiteSpace: 'nowrap',
        }}
      >
        {unsafe ? '历史附件链接已禁用' : displayName}
      </span>
      {protectedRef && <span style={{ color: 'var(--text-tertiary)' }}>受控</span>}
      {!readOnly && (
        <button onClick={onRemove} title="移除" style={{ border: 'none', background: 'none', color: 'var(--text-secondary)', cursor: 'pointer', display: 'flex' }}>
          <X size={14} />
        </button>
      )}
    </div>
  );
}

'use client';
import { useRef, useState } from 'react';
import { Paperclip, X, Loader2, FileText, ImageIcon } from 'lucide-react';

interface FileUploadInputProps {
  value?: string;
  onChange: (url: string) => void;
  disabled?: boolean;
  label?: string;
  /** 用于图片缩略图预览的基础 URL（留空则使用 NEXT_PUBLIC_API_BASE） */
  apiBase?: string;
}

function getToken() {
  if (typeof window === 'undefined') return '';
  return sessionStorage.getItem('phoenix_token') || localStorage.getItem('phoenix_token') || '';
}

function getApiBase() {
  if (typeof process !== 'undefined' && process.env.NEXT_PUBLIC_API_BASE) {
    return process.env.NEXT_PUBLIC_API_BASE;
  }
  return '';
}

function isUnsafeAttachmentRef(ref: string) {
  const value = ref.trim().toLowerCase();
  return value.startsWith('/uploads/') || value.startsWith('uploads/') || value.startsWith('http://') || value.startsWith('https://');
}

function isProtectedFileRef(ref: string) {
  return /^FILE-[A-Za-z0-9._:-]+$/.test(ref.trim());
}

export function FileUploadInput({
  value,
  onChange,
  disabled = false,
  label = '上传附件（PDF / 图片，≤15MB）',
  apiBase,
}: FileUploadInputProps) {
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState('');
  const inputRef = useRef<HTMLInputElement>(null);
  const base = apiBase ?? getApiBase();

  async function handleFile(file: File) {
    setUploading(true);
    setError('');
    try {
      const fd = new FormData();
      fd.append('file', file);
      const res = await fetch(`${base}/api/upload/contract`, {
        method: 'POST',
        headers: { Authorization: `Bearer ${getToken()}` },
        body: fd,
      });
      const json = await res.json().catch(() => ({}));
      if (json.code === 200 && json.fileId) {
        onChange(String(json.fileId));
      } else if (json.code === 200 && json.url) {
        onChange(String(json.url));
      } else {
        setError(json.message || '上传失败：服务端未返回受控 fileId');
      }
    } catch {
      setError('网络错误，请检查连接');
    } finally {
      setUploading(false);
    }
  }

  const unsafeRef = value ? isUnsafeAttachmentRef(value) : false;
  const protectedRef = value ? isProtectedFileRef(value) : false;
  const fileName = value
    ? protectedRef
      ? value
      : unsafeRef
        ? '历史附件链接已禁用'
        : decodeURIComponent(value.split('/').pop() || value)
    : '';

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
      <input
        ref={inputRef}
        type="file"
        accept=".pdf,.jpg,.jpeg,.png,.webp"
        style={{ display: 'none' }}
        onChange={e => {
          const f = e.target.files?.[0];
          if (f) handleFile(f);
          e.target.value = '';
        }}
      />

      {value ? (
        <div style={{
          display: 'flex', alignItems: 'center', gap: 8,
          padding: '6px 10px',
          border: '1px solid var(--border)',
          borderRadius: 'var(--radius-sm, 6px)',
          background: 'var(--bg-page)',
          fontSize: 13,
        }}>
          {protectedRef
            ? <FileText size={14} color="var(--color-accent)" style={{ flexShrink: 0 }} />
            : <ImageIcon size={14} color={unsafeRef ? 'var(--status-warning)' : 'var(--status-info)'} style={{ flexShrink: 0 }} />
          }
          <span
            title={unsafeRef ? '安全策略禁止打开裸链接' : fileName}
            style={{
              flex: 1,
              color: unsafeRef ? 'var(--status-warning)' : 'var(--color-primary)',
              overflow: 'hidden',
              textOverflow: 'ellipsis',
              whiteSpace: 'nowrap',
              fontSize: 12,
            }}
          >
            {fileName || '附件'}
          </span>
          {protectedRef && <span style={{ fontSize: 11, color: 'var(--text-tertiary)', flexShrink: 0 }}>受控</span>}
          {!disabled && (
            <>
              <button
                type="button"
                onClick={() => inputRef.current?.click()}
                disabled={uploading}
                style={{
                  background: 'none',
                  border: '1px solid var(--border)',
                  borderRadius: 4,
                  cursor: 'pointer',
                  fontSize: 11,
                  padding: '2px 7px',
                  color: 'var(--text-secondary)',
                  flexShrink: 0,
                }}
              >
                替换
              </button>
              <button
                type="button"
                onClick={() => onChange('')}
                title="移除附件"
                style={{
                  background: 'none',
                  border: 'none',
                  cursor: 'pointer',
                  color: 'var(--status-danger)',
                  padding: 2,
                  flexShrink: 0,
                  lineHeight: 1,
                }}
              >
                <X size={13} />
              </button>
            </>
          )}
        </div>
      ) : (
        <button
          type="button"
          onClick={() => !disabled && !uploading && inputRef.current?.click()}
          disabled={uploading || disabled}
          style={{
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            gap: 6,
            padding: '8px 14px',
            border: '1.5px dashed var(--border)',
            borderRadius: 'var(--radius-sm, 6px)',
            background: 'var(--bg-page)',
            cursor: uploading || disabled ? 'not-allowed' : 'pointer',
            fontSize: 13,
            color: 'var(--text-secondary)',
            width: '100%',
            opacity: disabled ? 0.55 : 1,
            transition: 'border-color 0.15s',
          }}
        >
          {uploading
            ? <Loader2 size={14} style={{ animation: 'spin 1s linear infinite' }} />
            : <Paperclip size={14} />
          }
          {uploading ? '上传中...' : label}
        </button>
      )}

      {error && (
        <span style={{ fontSize: 12, color: 'var(--status-danger)', marginTop: 2 }}>{error}</span>
      )}

      {value && (
        <span style={{ fontSize: 12, color: unsafeRef ? 'var(--status-warning)' : 'var(--text-tertiary)', marginTop: 2 }}>
          {unsafeRef ? '安全策略禁止裸链接访问，请重新上传生成受控附件引用。' : '附件已保存为受控引用，预览/下载需由业务模块受控端点提供。'}
        </span>
      )}

      <style>{`@keyframes spin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }`}</style>
    </div>
  );
}
